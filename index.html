import { GameEngine } from './gameEngine'; import { VibhattiType, VacanaType, WordData, CheckResult } from '../types'; export class UIManager { private engine: GameEngine; private selectedCardId: string | null = null; // Screens private startScreenEl!: HTMLElement; private playScreenEl!: HTMLElement; private resultsScreenEl!: HTMLElement; // Header / Stats Elements private currentWordTitleEl!: HTMLElement; private currentWordDescEl!: HTMLElement; private currentTimerEl!: HTMLElement; private currentScoreEl!: HTMLElement; // Game Board & Cards Area private boardGridEl!: HTMLElement; private sourceCardsContainerEl!: HTMLElement; // Result Elements private finalScoreEl!: HTMLElement; private feedbackMessageEl!: HTMLElement; private reviewListEl!: HTMLElement; constructor(engine: GameEngine) { this.engine = engine; } public init(): void { this.bindDOMElements(); this.setupEventListeners(); this.showScreen('start-screen'); } private bindDOMElements(): void { // Screens this.startScreenEl = document.getElementById('start-screen')!; this.playScreenEl = document.getElementById('play-screen')!; this.resultsScreenEl = document.getElementById('results-screen')!; // Info & Stats this.currentWordTitleEl = document.getElementById('current-word-title')!; this.currentWordDescEl = document.getElementById('current-word-desc')!; this.currentTimerEl = document.getElementById('current-timer')!; this.currentScoreEl = document.getElementById('current-score')!; // Containers this.boardGridEl = document.getElementById('board-grid')!; this.sourceCardsContainerEl = document.getElementById('source-cards-container')!; // Results this.finalScoreEl = document.getElementById('final-score')!; this.feedbackMessageEl = document.getElementById('feedback-message')!; this.reviewListEl = document.getElementById('review-list')!; } private setupEventListeners(): void { // หน้าแรก -> เริ่มเล่นเกม document.getElementById('btn-start')?.addEventListener('click', () => { this.engine.startNewGame(); this.showScreen('play-screen'); this.refreshUI(); }); // ปุ่มรีเซ็ตกระดาน document.getElementById('btn-reset')?.addEventListener('click', () => { this.engine.resetCurrentBoard(); this.refreshUI(); }); // ปุ่มตรวจคำตอบ document.getElementById('btn-submit')?.addEventListener('click', () => { const result = this.engine.checkAnswers(); this.renderResultsScreen(result); this.showScreen('results-screen'); }); // ปุ่มเล่นคำถัดไป document.getElementById('btn-restart')?.addEventListener('click', () => { this.engine.nextStage(); this.showScreen('play-screen'); this.refreshUI(); }); // ปุ่มกลับหน้าแรก document.getElementById('btn-home')?.addEventListener('click', () => { this.showScreen('start-screen'); }); } private showScreen(screenId: 'start-screen' | 'play-screen' | 'results-screen'): void { this.startScreenEl.classList.remove('active'); this.playScreenEl.classList.remove('active'); this.resultsScreenEl.classList.remove('active'); if (screenId === 'start-screen') this.startScreenEl.classList.add('active'); if (screenId === 'play-screen') this.playScreenEl.classList.add('active'); if (screenId === 'results-screen') this.resultsScreenEl.classList.add('active'); } private refreshUI(): void { const currentWord = this.engine.getCurrentWord(); if (!currentWord) return; // อัปเดตข้อมูลคำศัพท์ & คะแนน this.currentWordTitleEl.textContent = `ศัพท์หลัก: ${currentWord.baseWord}`; this.currentWordDescEl.textContent = `คำแปล: ${currentWord.translation} (${currentWord.declensionType})`; this.currentScoreEl.textContent = this.engine.getScore().toString(); this.currentTimerEl.textContent = this.engine.getElapsedTime().toFixed(1); this.selectedCardId = null; this.renderGrid(currentWord); this.renderSourceCards(); } private renderGrid(word: WordData): void { this.boardGridEl.innerHTML = ''; const vibhattis: VibhattiType[] = ['ปฐมา', 'ทุติยา', 'ตติยา', 'จตุตถี', 'ปัญจมี', 'ฉัฏฐี', 'สัตตมี', 'อาลปนะ']; const vacanas: VacanaType[] = ['เอกวจนะ', 'พหุวจนะ']; const placements = this.engine.getPlacements(); // สร้าง Header ของตาราง const headerRow = document.createElement('div'); headerRow.className = 'grid-header-row'; headerRow.innerHTML = `
วิภัตติ
เอกวจนะ
พหุวจนะ
`; this.boardGridEl.appendChild(headerRow); // สร้างแถวสำหรับแต่วิภัตติ vibhattis.forEach((vib) => { const rowEl = document.createElement('div'); rowEl.className = 'grid-row'; // Label วิภัตติ const labelCell = document.createElement('div'); labelCell.className = 'grid-label-cell'; labelCell.textContent = vib; rowEl.appendChild(labelCell); // ช่องใส่การ์ด เอกวจนะ / พหุวจนะ vacanas.forEach((vac) => { const slotKey = `${vib}_${vac}`; const slotEl = document.createElement('div'); slotEl.className = 'grid-slot-cell'; slotEl.dataset.vibhatti = vib; slotEl.dataset.vacana = vac; const placedData = placements.get(slotKey); if (placedData && placedData.cards.length > 0) { placedData.cards.forEach((card) => { const cardEl = this.createCardElement(card.id, card.text, true); slotEl.appendChild(cardEl); }); } else { slotEl.innerHTML = 'วางการ์ดที่นี่'; } // Drag & Drop Events slotEl.addEventListener('dragover', (e) => e.preventDefault()); slotEl.addEventListener('drop', (e) => { e.preventDefault(); const cardId = e.dataTransfer?.getData('text/plain'); const cardText = e.dataTransfer?.getData('card-text'); if (cardId && cardText) { this.engine.placeCard(vib, vac, cardId, cardText); this.refreshUI(); } }); // Click to Place slotEl.addEventListener('click', () => { if (this.selectedCardId) { const card = this.engine.getDeck().find((c) => c.id === this.selectedCardId); if (card) { this.engine.placeCard(vib, vac, card.id, card.text); this.refreshUI(); } } }); rowEl.appendChild(slotEl); }); this.boardGridEl.appendChild(rowEl); }); } private renderSourceCards(): void { this.sourceCardsContainerEl.innerHTML = ''; const unplacedCards = this.engine.getUnplacedCards(); unplacedCards.forEach((card) => { const cardEl = this.createCardElement(card.id, card.text, false); this.sourceCardsContainerEl.appendChild(cardEl); }); } private createCardElement(id: string, text: string, isPlaced: boolean): HTMLElement { const cardEl = document.createElement('div'); cardEl.className = `card ${isPlaced ? 'placed-card' : ''} ${this.selectedCardId === id ? 'selected' : ''}`; cardEl.textContent = text; cardEl.draggable = true; // Drag Start cardEl.addEventListener('dragstart', (e) => { e.dataTransfer?.setData('text/plain', id); e.dataTransfer?.setData('card-text', text); }); // Click Selection / Remove cardEl.addEventListener('click', (e) => { e.stopPropagation(); if (isPlaced) { // ถ้านำออกจากตาราง this.engine.removeCardById(id); this.refreshUI(); } else { // คลิกเลือกเพื่อย้าย this.selectedCardId = this.selectedCardId === id ? null : id; this.renderSourceCards(); } }); return cardEl; } private renderResultsScreen(result: CheckResult): void { this.finalScoreEl.textContent = `${result.score} / ${result.total}`; // แสดงข้อความ Feedback const percentage = (result.score / result.total) * 100; if (percentage === 100) { this.feedbackMessageEl.textContent = 'ยอดเยี่ยมมาก! คุณตอบถูกต้องทั้งหมด 🎉'; } else if (percentage >= 70) { this.feedbackMessageEl.textContent = 'เก่งมาก! ทำได้ดีแล้ว ลองทบทวนจุดที่ผิดอีกนิดนะ 👍'; } else { this.feedbackMessageEl.textContent = 'พยายามอีกนิด! ลองศึกษาเฉลยแล้วลองใหม่อีกครั้งนะ ✌️'; } // รายการทบทวนคำตอบ this.reviewListEl.innerHTML = ''; result.results.forEach((r) => { const item = document.createElement('div'); item.className = `review-item ${r.isCorrect ? 'correct' : 'wrong'}`; item.innerHTML = `
${r.vibhatti} (${r.vacana}): ${r.isCorrect ? 'ถูกต้อง ✅' : 'ผิด ❌'}

คำตอบของคุณ: ${r.userAnswers.join(', ') || '(ไม่ได้วางการ์ด)'}

${!r.isCorrect ? `

เฉลยที่ถูกต้อง: ${r.correctAnswers.join(', ')}

` : ''}
`; this.reviewListEl.appendChild(item); }); } }
